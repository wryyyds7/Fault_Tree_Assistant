三、前端API调用结构 (src/api/index.js)
================================================================================

  基础配置：
  - baseURL: '/api/v1'
  - timeout: 30000ms
  - 自动携带 JWT Token (Authorization: Bearer {token})
  - 统一错误处理

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  1. 认证相关 (auth)                                                             │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /auth/login           用户登录                                          │
  │  POST   /auth/register        用户注册                                          │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  2. 文档管理 (document)                                                         │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /documents/upload              上传文档 (multipart/form-data)           │
  │  GET    /documents/{docId}            获取文档元数据                            │
  │  GET    /documents/{docId}/content    获取文档内容                              │
  │  GET    /documents                    获取文档列表                               │
  │  DELETE /documents/{docId}            删除文档                                  │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  3. 故障树管理 (faultTree)                                                       │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  GET    /fault-trees                 获取故障树列表 (分页/过滤)                   │
  │  GET    /fault-trees/{treeId}        获取故障树详情                              │
  │  POST   /fault-trees                 创建故障树                                  │
  │  PUT    /fault-trees/{treeId}        更新故障树                                 │
  │  DELETE /fault-trees/{treeId}        删除故障树                                 │
  │                                                                                 │
  │  节点操作：                                                                      │
  │  POST   /fault-trees/{treeId}/nodes                           添加节点           │
  │          参数: parentEventId (query)                                        │
  │  PUT    /fault-trees/{treeId}/nodes                           更新节点           │
  │  DELETE /fault-trees/{treeId}/nodes/{eventId}              删除节点           │
  │  PUT    /fault-trees/{treeId}/nodes/{eventId}/move          移动节点           │
  │          参数: newParentId (query)                                          │
  │  PUT    /fault-trees/{treeId}/positions                    更新节点位置       │
  │  PUT    /fault-trees/{treeId}/nodes/{eventId}/confidence   更新置信度         │
  │          Body: { confidence, verificationStatus }                            │
  │  PUT    /fault-trees/{treeId}/nodes/{eventId}/verification 更新验证状态       │
  │          Body: { status }                                                   │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  4. 规则校验 (validation)                                                       │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /validation/validate    校验故障树                                      │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  5. RAG生成 (rag)                                                               │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /rag/generate                 生成故障树任务                              │
  │          Body: { description, equipmentType, knowledgeDriven, dataDriven }    │
  │  GET    /rag/tasks/{taskId}           获取任务状态                               │
  │  GET    /rag/evidence/{paragraphId}  获取证据段落                               │
  │  POST   /rag/evidence                 批量获取证据                              │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  6. 知识图谱 (knowledgeGraph)                                                    │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /kg/query-template   查询模板 (三层匹配)                                │
  │  PUT    /kg/enrich           知识丰富化                                        │
  │  POST   /kg/initialize       初始化本体                                        │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  7. 向量存储 (vector)                                                            │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /vector/search                  向量相似度搜索                           │
  │  POST   /vector/search-with-evidence   带证据搜索                               │
  │  GET    /vector/paragraphs/{paragraphId}/evidence  获取段落证据                 │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  8. 反馈学习 (feedback)                                                          │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  POST   /feedback                        创建反馈                               │
  │  GET    /feedback/{feedbackId}           获取反馈详情                           │
  │  GET    /feedback/tree/{treeId}          按故障树获取反馈                       │
  │  GET    /feedback/user/{userId}          按用户获取反馈                         │
  │  GET    /feedback                        获取所有反馈                           │
  │  POST   /feedback/process-batch          批量处理反馈                           │
  │  POST   /feedback/optimize-models       优化模型                               │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  ┌─────────────────────────────────────────────────────────────────────────────────┐
  │  9. 统计 (stats)                                                                │
  ├─────────────────────────────────────────────────────────────────────────────────┤
  │                                                                                 │
  │  GET    /stats/dashboard              获取仪表盘统计                             │
  │                                                                                 │
  └─────────────────────────────────────────────────────────────────────────────────┘

  DELETE   /api/v1/fault-trees/{treeId}        fault-tree-editor    删除树
